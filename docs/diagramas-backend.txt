' ============================================================
package "View (Camada de Servicos)" <<Folder>> {

  class "POST /api/sync" as SyncRoute <<route>> {
    + POST(request) : Response
  }
  class "GET|PUT /api/sync/[code]" as SyncCodeRoute <<route>> {
    + GET(request, params) : Response
    + PUT(request, params) : Response
  }
  class "DELETE /api/sync/[code]/delete" as DeleteRoute <<route>> {
    + DELETE(request, params) : Response
  }
  class "GET|POST /api/sync/[code]/password" as PasswordRoute <<route>> {
    + GET(request, params) : Response
    + POST(request, params) : Response
  }
  class "GET /api/profiles" as ProfilesRoute <<route>> {
    + GET(request) : Response
  }
  class "GET /api/stats" as StatsRoute <<route>> {
    + GET() : Response
  }
}

    + {abstract} index(page, limit) : Promise

    + {abstract} destroy(code, password) : Promise
    + {abstract} search(query) : Promise

    - dao : IProfileDAO
    + constructor(dao : IProfileDAO)
    - generateSyncCode() : string
    + index(page, limit) : Promise
    + show(code) : Promise
    + store(data) : Promise
    + update(code, data) : Promise
    + destroy(code, password) : Promise
    + search(query) : Promise
    + setPassword(code, password) : Promise
    + hasPassword(code) : Promise
    + stats() : Promise
  }

  class "config (Factory + DI)" as Config <<factory>> {

    - createDAO() : IProfileDAO
    - createController() : IProfileController
    + profileController : IProfileController
  }
}

@startuml SequenciaCriarPerfil

actor Cliente
participant "POST /api/sync\n(View)" as Route
participant "profileController\n(Config/Factory)" as Config
participant "ProfileController\n(Controller)" as Ctrl
participant "ProfileDAO_Neon\n(Persistencia)" as DAO
database "Neon\n(sync_profiles)" as DB

Cliente -> Route : POST { icons, settings, isWelcomeComplete }
activate Route
Route -> Route : valida (icons e settings presentes)
Route -> Config : profileController.store(data)
activate Config
Config -> Ctrl : store(data)
activate Ctrl
Ctrl -> Ctrl : generateSyncCode()
Ctrl -> DAO : create({ syncCode, stateData })
activate DAO
DAO -> DB : INSERT INTO sync_profiles ... RETURNING *
activate DB
DB --> DAO : registro criado
deactivate DB
DAO --> Ctrl : registro
deactivate DAO
Ctrl --> Config : { syncCode }
deactivate Ctrl
Config --> Route : { syncCode }
deactivate Config
Route --> Cliente : 201 Created { syncCode }
deactivate Route

@enduml

 3. DIAGRAMA DE SEQUENCIA - Buscar Perfil (GET /api/sync/[code] -> show)
================================================================================

@startuml SequenciaBuscarPerfil
title Sequencia - Buscar Perfil (GET /api/sync/[code])

actor Cliente
participant "GET /api/sync/[code]\n(View)" as Route
participant "profileController" as Config
participant "ProfileController" as Ctrl
participant "ProfileDAO_Neon" as DAO
database "Neon" as DB

Cliente -> Route : GET /api/sync/{code}
activate Route
Route -> Route : code.toUpperCase() + valida regex
alt codigo invalido
  Route --> Cliente : 400 Codigo invalido
else codigo valido
  Route -> Config : profileController.show(code)
  activate Config
  Config -> Ctrl : show(code)
  activate Ctrl
  Ctrl -> DAO : recoveryByCode(code)
  activate DAO
  DAO -> DB : SELECT ... WHERE sync_code = code
  activate DB
  DB --> DAO : registro ou null
  deactivate DB
  DAO --> Ctrl : registro
  deactivate DAO
  alt perfil nao encontrado
    Ctrl --> Config : null
    Config --> Route : null
    Route --> Cliente : 404 Perfil nao encontrado
  else perfil encontrado
    Ctrl --> Config : { syncCode, stateData, hasPassword, ... }
    deactivate Ctrl
    Config --> Route : profile
    deactivate Config
    Route --> Cliente : 200 { stateData, updatedAt }
  end
end
deactivate Route

@enduml

 5. DIAGRAMA DE SEQUENCIA - Deletar Perfil com Senha
    (DELETE /api/sync/[code]/delete -> hasPassword + destroy)
================================================================================

@startuml SequenciaDeletarPerfil
title Sequencia - Deletar Perfil (DELETE /api/sync/[code]/delete)

actor Cliente
participant "DELETE .../delete\n(View)" as Route
participant "profileController" as Config
participant "ProfileController" as Ctrl
participant "ProfileDAO_Neon" as DAO
participant "bcryptjs" as Bcrypt
database "Neon" as DB

Cliente -> Route : DELETE { password? }
activate Route
Route -> Config : profileController.hasPassword(code)
activate Config
Config -> Ctrl : hasPassword(code)
activate Ctrl
Ctrl -> DAO : getPasswordHash(code)
activate DAO
DAO -> DB : SELECT delete_password WHERE sync_code = code
DB --> DAO : hash ou null
DAO --> Ctrl : hash
deactivate DAO
Ctrl --> Config : boolean (!!hash)
deactivate Ctrl
Config --> Route : hasPassword
deactivate Config

alt tem senha e nenhuma senha enviada
  Route --> Cliente : 401 Senha obrigatoria (requiresPassword)
else prosseguir exclusao
  Route -> Config : profileController.destroy(code, password)
  activate Config
  Config -> Ctrl : destroy(code, password)
  activate Ctrl
  Ctrl -> DAO : recoveryByCode(code)
  activate DAO
  DAO -> DB : SELECT ... WHERE sync_code = code
  DB --> DAO : registro
  DAO --> Ctrl : registro
  deactivate DAO
  alt registro tem delete_password
    Ctrl -> Bcrypt : compare(password, delete_password)
    activate Bcrypt
    Bcrypt --> Ctrl : valido?
    deactivate Bcrypt
    alt senha invalida
      Ctrl --> Config : false
      Config --> Route : false
      Route --> Cliente : 403 Senha incorreta
    else senha valida
      Ctrl -> DAO : delete(code)
      activate DAO
      DAO -> DB : DELETE FROM sync_profiles WHERE sync_code = code
      DB --> DAO : linhas afetadas
      DAO --> Ctrl : boolean
      deactivate DAO
      Ctrl --> Config : true
      Config --> Route : true
      Route --> Cliente : 200 Perfil deletado
    end
  else sem senha (perfil livre)
    Ctrl -> DAO : delete(code)
    activate DAO
    DAO -> DB : DELETE FROM sync_profiles WHERE sync_code = code
    DB --> DAO : boolean
    DAO --> Ctrl : boolean
    deactivate DAO
    Ctrl --> Config : true
    Config --> Route : true
    Route --> Cliente : 200 Perfil deletado
  end
  deactivate Ctrl
  deactivate Config
end
deactivate Route

@enduml

 6. DIAGRAMA DE SEQUENCIA - Definir Senha
    (POST /api/sync/[code]/password -> show + setPassword)
================================================================================

@startuml SequenciaDefinirSenha
title Sequencia - Definir Senha (POST /api/sync/[code]/password)

actor Cliente
participant "POST .../password\n(View)" as Route
participant "profileController" as Config
participant "ProfileController" as Ctrl
participant "ProfileDAO_Neon" as DAO
participant "bcryptjs" as Bcrypt
database "Neon" as DB

Cliente -> Route : POST { password }
activate Route
Route -> Route : valida senha (string, <= 20 chars)
Route -> Config : profileController.show(code)
activate Config
Config -> Ctrl : show(code)
activate Ctrl
Ctrl -> DAO : recoveryByCode(code)
DAO -> DB : SELECT ... WHERE sync_code = code
DB --> DAO : registro
DAO --> Ctrl : registro
Ctrl --> Config : profile (hasPassword)
deactivate Ctrl
Config --> Route : profile
deactivate Config

alt perfil inexistente
  Route --> Cliente : 404 Perfil nao encontrado
else ja possui senha
  Route --> Cliente : 400 Perfil ja possui senha
else definir senha
  Route -> Config : profileController.setPassword(code, password)
  activate Config
  Config -> Ctrl : setPassword(code, password)
  activate Ctrl
  Ctrl -> Bcrypt : hash(password, 10)
  activate Bcrypt
  Bcrypt --> Ctrl : passwordHash
  deactivate Bcrypt
  Ctrl -> DAO : setPassword(code, passwordHash)
  activate DAO
  DAO -> DB : UPDATE sync_profiles SET delete_password ... RETURNING *
  DB --> DAO : linhas afetadas
  DAO --> Ctrl : boolean
  deactivate DAO
  Ctrl --> Config : boolean
  deactivate Ctrl
  Config --> Route : success
  deactivate Config
  Route --> Cliente : 200 Senha definida com sucesso
end
deactivate Route

@enduml

DAO -> DB : SELECT ... ORDER BY created_at ASC LIMIT 1
DB --> DAO : resultados
DAO --> Ctrl : valores agregados

Ctrl -> Ctrl : monta recentProfiles + oldestProfile
Ctrl --> Config : { totalProfiles, today, week, recent, oldest }
deactivate Ctrl

deactivate Config
